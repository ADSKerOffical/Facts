Ключевые слова
   В C++ очень много ключевых слов. Они являются как дополнительными командами для компилятора
   Их более 80 штук. Вот их список:
   
      • auto – автоматически подбирает тип к переменной
         Его нельзя использовать: для нестатических членов класса, массивах, для неинициализированных переменных, в операторахведения типов (по типу sizeof(auto);), в возвращаемом значении функции (без хвостового типа или тела)
          
          auto a = 1; // int
          auto b = "Hello"; // const char*
          auto const pi = 3.1415f; // const float
          int arr[] = {1, 2, 3};
          auto x = arr; // x будет иметь тип int*, а не массив int[3]
      
      • const – константа (переменную нельзя изменить)
         Его нельзя использовать: конструкторы и деструкторы классов (например const MyClass), статические методы (например static void foo() const)
      
         const char* a = "Text";
         void able() const { // это означает, что этот метод класса не имеет права изменять поля (члены) объекта, для которого он вызван
            std::cout << a << std::endl;
         }
      
      • if, else, elseif – условные операторы
      
          if (1 + 2 == 4) {
             std::cout << "Это правильно" << std::endl;
          } else {
             std::cout << "Это не правильно" << std::endl;
          }
      
      • switch, case, default – операторы выбора
      • break – останавливает цикл
      
        while (i <= 5) {
           i += 1;
           std::cout << i << std::endl;
           if (i >= 5) {
              break;
           }
         }
      
      • continue – переход к следующей итерации цикла
      • try, catch, finally, throw – защита от ошибок
      • goto – безусловный переход
         В современном C++ использовать goto для создания обычных циклов крайне не рекомендуется. Это ухудшает читаемость кода (создает так называемый «спагетти-код»). Стандартные циклы while или for справляются с этой задачей гораздо лучше. Единственный редкий случай, когда goto оправдан — это быстрый выход из глубоко вложенных циклов
      
         int main() {
            int value = 1;
            
            loop_start;
            
            std::cout << value << " ";
            value++;
            
            if (value <= 5) {
               goto loop_start;
            }
            std::cout << "Конец" << std::endl;
            return 0;
         }
         
      • constexpr – вычисление на этапе компиляции
      • mutable – изменяемый член внутри const объекта
      • return – то что должна вернуть функция
        
        int abc(int value) {
           return value * 2;
        }
        int value2 = abc(4); // 8
        
      • this – указатель на текущий объект класса
      • using – объявление псевдонима типа или пространства имен
      
       #include <iostream>
       #include <vector>
       #include <string>

        // создание более короткого и понятного псевдонима для сложного типа
        using StringList = std::vector<std::string>;

        int main() {
          // чтобы не писать std::cout каждый раз
          using std::cout;
          using std::endl;

          StringList fruits = {"Яблоко", "Банан", "Апельсин"};
          for (const auto& fruit : fruits) {
             cout << fruit << endl; 
           }
          return 0;
         }

      • thread_local – переменная внутри текущего потока
      • typedef – создание синонима типа
      • virtual – виртуальная функция
      • friend – дружественная функция или класс
      • public, private, protected – спецификаторы доступа
      • const_cast – снятие или добавления const
         Ключевое слово снимает константность только с указателей и ссылок. Если сам исходный объект был объявлен как const, то попытка изменить его через const_cast приведет к неопределенному поведению (UB), так как компилятор мог поместить эту переменную в память только для чтения
         
      • static_cast – статическое приведение типов
      • union – объедение
      • enum – перечисление
