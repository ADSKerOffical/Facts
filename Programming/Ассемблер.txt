• Ассемблер
     Ассемблер – это язык программирования низкого уровня, в котором команды заменены на короткие слова (мнемоники) для удобства человека, а также программа-транслятор, которая переводит этот код в понятные процессору нули и единицы (машинный код). Он считается самым низкоуровневым языком, не считая машинный код
     Для каждой архитектуры нужно писать конкретный код, иначе устройство не поймёт как правильно запустить код
     Чтобы он заработал его для начала нужно скомпилировать, например NASM, GAS, FASM, MASM. Если запустить на основной процессор и допустить ошибку, то может вывести синий экран смерти или kernel panic
     Расширение файлов у ассемблера это: asm, s, as, inc
     Для примера вот так выглядит код для написания Hello World на архитектуре aarch64 и ОС Linux:
     
.global _start          // Точка входа для линковщика

.section .data
msg:
    .ascii "Hello, World!\n"  // Текст сообщения
    len = . - msg             // Вычисление длины строки

.section .text
_start:
    // Системный вызов write (номер 64)
    mov x0, #1          // Дескриптор файла 1 (stdout - экран)
    adr x1, msg         // Адрес строки сообщения
    mov x2, #len        // Длина сообщения
    mov w8, #64         // Номер системного вызова write в Linux ARM64
    svc #0              // Вызов ядра ОС

    // Системный вызов exit (номер 93)
    mov x0, #0          // Код возврата 0 (успех)
    mov w8, #93         // Номер системного вызова exit в Linux ARM64
    svc #0              // Завершение программы
